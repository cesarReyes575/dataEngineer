CREATE OR REPLACE VIEW vistaCompany AS
SELECT
    DATE_TRUNC('day',ch.created_at) AS dia,
	co.company_name,
    SUM (ch.amount) AS suma_total       
FROM
    charges ch
INNER JOIN 
	companies co ON ch.company_id = co.company_id
GROUP BY
    ch.created_at,
	co.company_name
ORDER BY dia ASC;

SELECT * FROM vistaCompany;
